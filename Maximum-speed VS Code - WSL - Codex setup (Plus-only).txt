One epic game concept: The Wastes Courier: Curator of the Void
You’re a courier/curator hauling cursed artefacts across a ruined world to rebuild a legendary Museum, and each “job” drops you into a different roguelike layer:
The 3-layer game structure (this is what makes it work)
1. Overworld (Courier of the Wastes)

   * Procedural route map of towns + danger zones
   * You pick contracts: deliver artefact, retrieve relic, escort cargo
   * Resources: fuel, supplies, time/heat, reputation
   2. Dungeons (the roguelike runs) Each node on the overworld can be one of these:

      * Derelict Starship (Salvage): oxygen/time pressure, doors lock, systems reboot, escape pods as exits
      * Wizard Tower (Spellcraft): rune rooms, spell-combo trials, arcane hazards
      * Hive Nest (Summoner): swarm tactics, protect the “core,” evolve minions mid-run
      3. Home Base (Museum of Curses)

         * Your “stash” is the Museum vault
         * Every artefact is boon + curse, and displaying them changes the world rules
         * Meta progression = restoring wings of the Museum (unlocks new routes, vendors, starting loadouts)
The glue that unifies everything
1) Artefacts = your build
         * Artefacts are the main loot everywhere.
         * Each has:
         * Boon (damage type, movement, summon buff, economy)
         * Curse (map distortion, enemy mutators, inventory constraints, “reality events”)
2) Runes = your spell system
         * Runes drop mostly in the Tower, but also rare in ships/hives.
         * You craft spells as: Form + Element + Modifier
         * Form: bolt / beam / trap / aura / summon
         * Element: fire / ice / void / shock / rot
         * Modifier: chain / pierce / blink / leech / split
3) Minions = drones/swarm (fits all biomes)
         * In ships: they’re salvage drones
         * In hives: they’re bio-minions
         * In towers: they’re arcane constructs Mechanically it’s the same system, just reskinned per biome.
Win condition that feels “multi-goal”
You’re trying to complete a Grand Delivery: assemble a set of keystone artefacts and survive the final route to return them to the Museum. Each biome gives a different “keystone” and the curses stack, so by the end the run is chaotic in a fun way.

Then you “swap in”:

import Phaser from "phaser";
import { OverworldScene } from "./scenes/OverworldScene";
import { RunScene } from "./scenes/RunScene";
import { MuseumScene } from "./scenes/MuseumScene";


export function createGame(parent: string) {
  const config: Phaser.Types.Core.GameConfig = {
    type: Phaser.AUTO,
    parent,
    backgroundColor: "#0b0f14",
    scene: [OverworldScene, RunScene, MuseumScene],
    scale: {
      mode: Phaser.Scale.FIT,
      autoCenter: Phaser.Scale.CENTER_BOTH,
      width: 960,
      height: 540,
    },
    render: {
      pixelArt: true,
      antialias: false,
    },
  };


  return new Phaser.Game(config);
}

1) Replace src/game/scenes/RunScene.ts with this
import Phaser from "phaser";
import { SceneKeys } from "../SceneKeys";


type RunData = { entry?: string };


type ActorKind = "player" | "enemy";
type Actor = {
  id: string;
  kind: ActorKind;
  x: number; // tile x
  y: number; // tile y
  rect: Phaser.GameObjects.Rectangle;
};


type Tile = 0 | 1; // 0 floor, 1 wall


export class RunScene extends Phaser.Scene {
  // Grid config
  private readonly gridW = 20;
  private readonly gridH = 12;
  private readonly tileSize = 36;


  // Rendering/layout
  private originX = 0;
  private originY = 0;
  private gridGfx!: Phaser.GameObjects.Graphics;


  // Map + actors
  private tiles: Tile[] = [];
  private actors: Actor[] = [];
  private occupancy = new Map<string, string>(); // "x,y" -> actorId
  private turnIndex = 0;
  private phase: "PLAYER" | "AI_BUSY" = "PLAYER";


  // UI
  private turnText!: Phaser.GameObjects.Text;
  private helpText!: Phaser.GameObjects.Text;


  constructor() {
    super(SceneKeys.Run);
  }


  init(data: RunData) {
    this.registry.set("runEntry", data.entry ?? "unknown");
  }


  create() {
    const entry = this.registry.get("runEntry") as string;


    // Keep your scene switching keys
    this.input.keyboard?.on("keydown-O", () => this.scene.start(SceneKeys.Overworld));
    this.input.keyboard?.on("keydown-M", () => this.scene.start(SceneKeys.Museum, { entry: "run" }));


    // Layout: centre grid with room for header text
    const gridPxW = this.gridW * this.tileSize;
    const gridPxH = this.gridH * this.tileSize;


    this.originX = Math.floor((this.scale.width - gridPxW) / 2);
    this.originY = Math.floor((this.scale.height - gridPxH) / 2) + 30;


    this.add.text(20, 18, "RUN: STARSHIP (GRID MVP)", {
      fontSize: "22px",
      color: "#ffffff",
    });


    this.add.text(20, 44, `Entry: ${entry}`, {
      fontSize: "14px",
      color: "#bbbbbb",
    });


    this.turnText = this.add.text(20, 66, "", {
      fontSize: "14px",
      color: "#dddddd",
    });


    this.helpText = this.add.text(
      20,
      90,
      [
        "Move: Arrow keys / WASD",
        "Wait: Space",
        "Scene: [O] Overworld, [M] Museum",
        "",
        "Rule: you move 1 tile, then enemy moves 1 tile.",
      ].join("\n"),
      { fontSize: "14px", color: "#9fb0c0" }
    );


    // Build map + draw it once
    this.generateMap();
    this.gridGfx = this.add.graphics();
    this.drawGrid();


    // Spawn actors
    this.spawnActors();


    // Input: single-step movement on key press (not held)
    this.input.keyboard?.on("keydown", (ev: KeyboardEvent) => {
      if (this.phase !== "PLAYER") return;


      const key = ev.key.toLowerCase();
      const dir = this.keyToDir(key);
      if (!dir) return;


      const player = this.actors.find(a => a.kind === "player");
      if (!player) return;


      if (dir.type === "wait") {
        this.endPlayerTurn();
        return;
      }


      const moved = this.tryMove(player, dir.dx, dir.dy);
      if (moved) this.endPlayerTurn();
    });


    // Start turn queue
    this.turnIndex = 0;
    this.phase = "PLAYER";
    this.updateTurnText();
  }


  // ---------- Map ----------
  private idx(x: number, y: number) {
    return y * this.gridW + x;
  }


  private inside(x: number, y: number) {
    return x >= 0 && y >= 0 && x < this.gridW && y < this.gridH;
  }


  private isWall(x: number, y: number) {
    if (!this.inside(x, y)) return true;
    return this.tiles[this.idx(x, y)] === 1;
  }


  private setTile(x: number, y: number, t: Tile) {
    this.tiles[this.idx(x, y)] = t;
  }


  private generateMap() {
    // Base floor
    this.tiles = new Array(this.gridW * this.gridH).fill(0);


    // Border walls
    for (let x = 0; x < this.gridW; x++) {
      this.setTile(x, 0, 1);
      this.setTile(x, this.gridH - 1, 1);
    }
    for (let y = 0; y < this.gridH; y++) {
      this.setTile(0, y, 1);
      this.setTile(this.gridW - 1, y, 1);
    }


    // Random interior walls (light density)
    const density = 0.12;
    for (let y = 1; y < this.gridH - 1; y++) {
      for (let x = 1; x < this.gridW - 1; x++) {
        if (Math.random() < density) this.setTile(x, y, 1);
      }
    }


    // Ensure spawn/paths aren’t blocked
    this.setTile(1, 1, 0);
    this.setTile(2, 1, 0);
    this.setTile(1, 2, 0);


    this.setTile(this.gridW - 2, this.gridH - 2, 0);
    this.setTile(this.gridW - 3, this.gridH - 2, 0);
    this.setTile(this.gridW - 2, this.gridH - 3, 0);
  }


  private drawGrid() {
    this.gridGfx.clear();


    const floorFill = 0x121a24;
    const wallFill = 0x2a3340;
    const stroke = 0x223044;


    for (let y = 0; y < this.gridH; y++) {
      for (let x = 0; x < this.gridW; x++) {
        const wx = this.originX + x * this.tileSize;
        const wy = this.originY + y * this.tileSize;


        const isWall = this.tiles[this.idx(x, y)] === 1;
        this.gridGfx.fillStyle(isWall ? wallFill : floorFill, 1);
        this.gridGfx.fillRect(wx, wy, this.tileSize, this.tileSize);


        this.gridGfx.lineStyle(1, stroke, 1);
        this.gridGfx.strokeRect(wx, wy, this.tileSize, this.tileSize);
      }
    }
  }


  // ---------- Actors + Turn Queue ----------
  private key(x: number, y: number) {
    return `${x},${y}`;
  }


  private tileToWorldCenter(x: number, y: number) {
    return {
      x: this.originX + x * this.tileSize + this.tileSize / 2,
      y: this.originY + y * this.tileSize + this.tileSize / 2,
    };
  }


  private spawnActors() {
    this.actors = [];
    this.occupancy.clear();


    const spawn = (id: string, kind: ActorKind, x: number, y: number, color: number) => {
      const p = this.tileToWorldCenter(x, y);
      const rect = this.add
        .rectangle(p.x, p.y, this.tileSize * 0.72, this.tileSize * 0.72, color)
        .setStrokeStyle(2, 0x0b0f14);


      const actor: Actor = { id, kind, x, y, rect };
      this.actors.push(actor);
      this.occupancy.set(this.key(x, y), id);
      return actor;
    };


    // Player at (1,1), enemy at opposite corner
    spawn("player", "player", 1, 1, 0x52d6ff);
    spawn("enemy-1", "enemy", this.gridW - 2, this.gridH - 2, 0xff5c7a);
  }


  private actorAt(x: number, y: number): Actor | undefined {
    const id = this.occupancy.get(this.key(x, y));
    if (!id) return undefined;
    return this.actors.find(a => a.id === id);
  }


  private tryMove(actor: Actor, dx: number, dy: number) {
    const nx = actor.x + dx;
    const ny = actor.y + dy;


    if (!this.inside(nx, ny)) return false;
    if (this.isWall(nx, ny)) return false;


    const occ = this.actorAt(nx, ny);
    if (occ) return false; // later: attack/interact


    // Update occupancy map
    this.occupancy.delete(this.key(actor.x, actor.y));
    actor.x = nx;
    actor.y = ny;
    this.occupancy.set(this.key(actor.x, actor.y), actor.id);


    // Move visuals
    const p = this.tileToWorldCenter(actor.x, actor.y);
    actor.rect.setPosition(p.x, p.y);


    return true;
  }


  private endPlayerTurn() {
    // Advance to next actor
    this.turnIndex = (this.turnIndex + 1) % this.actors.length;
    this.updateTurnText();
    this.beginCurrentTurn();
  }


  private beginCurrentTurn() {
    const actor = this.actors[this.turnIndex];
    if (!actor) return;


    if (actor.kind === "player") {
      this.phase = "PLAYER";
      this.updateTurnText();
      return;
    }


    // AI turn (single-step)
    this.phase = "AI_BUSY";
    this.updateTurnText();


    this.time.delayedCall(120, () => {
      this.aiAct(actor);


      // End AI turn
      this.turnIndex = (this.turnIndex + 1) % this.actors.length;
      this.updateTurnText();


      // Chain to next (player)
      this.beginCurrentTurn();
    });
  }


  private aiAct(enemy: Actor) {
    const player = this.actors.find(a => a.kind === "player");
    if (!player) return;


    const dx = player.x - enemy.x;
    const dy = player.y - enemy.y;


    // Prefer larger axis; try alternate if blocked
    const stepX = Math.sign(dx);
    const stepY = Math.sign(dy);


    const preferX = Math.abs(dx) >= Math.abs(dy);


    const attempts: Array<[number, number]> = preferX
      ? [[stepX, 0], [0, stepY]]
      : [[0, stepY], [stepX, 0]];


    // If both are zero (same tile, shouldn’t happen), do nothing.
    for (const [ax, ay] of attempts) {
      if (ax === 0 && ay === 0) continue;
      if (this.tryMove(enemy, ax, ay)) return;
    }
    // If blocked, enemy waits.
  }


  private updateTurnText() {
    const actor = this.actors[this.turnIndex];
    const who = actor ? `${actor.kind.toUpperCase()} (${actor.id})` : "—";
    this.turnText.setText(`Turn: ${who}  |  Phase: ${this.phase}`);
  }


  // ---------- Input helpers ----------
  private keyToDir(key: string):
    | { type: "move"; dx: number; dy: number }
    | { type: "wait" }
    | null {
    if (key === " " || key === "space") return { type: "wait" };


    if (key === "arrowup" || key === "w") return { type: "move", dx: 0, dy: -1 };
    if (key === "arrowdown" || key === "s") return { type: "move", dx: 0, dy: 1 };
    if (key === "arrowleft" || key === "a") return { type: "move", dx: -1, dy: 0 };
    if (key === "arrowright" || key === "d") return { type: "move", dx: 1, dy: 0 };


    return null;
  }
}

```typescript
export class SeededRNG {
  private seed: number;


  constructor(seed: number) {
    this.seed = seed;
  }


  // Linear congruential generator
  next(): number {
    this.seed = (this.seed * 1664525 + 1013904223) % 4294967296;
    return this.seed / 4294967296;
  }


  nextInt(min: number, max: number): number {
    return Math.floor(this.next() * (max - min + 1)) + min;
  }


  nextFloat(min: number, max: number): number {
    return this.next() * (max - min) + min;
  }


  choice<T>(array: T[]): T {
    return array[this.nextInt(0, array.length - 1)];
  }


  getSeed(): number {
    return this.seed;
  }
}
```

  if (!exitSpawn) throw new Error(`Room ${room.id} missing exit E`);


  this.exitPos = exitSpawn;


  return { roomId: room.id, playerSpawn, enemySpawns };
}

import { GameConstants } from "../constants";


export type Tile = 0 | 1;


export class GridSystem {
  readonly w = GameConstants.GRID_WIDTH;
  readonly h = GameConstants.GRID_HEIGHT;


  tiles: Tile[] = new Array(this.w * this.h).fill(0);
  occupancy = new Map<string, string>(); // "x,y" => actorId


  idx(x: number, y: number) {
    return y * this.w + x;
  }


  key(x: number, y: number) {
    return `${x},${y}`;
  }


  inside(x: number, y: number) {
    return x >= 0 && y >= 0 && x < this.w && y < this.h;
  }


  isWall(x: number, y: number) {
    if (!this.inside(x, y)) return true;
    return this.tiles[this.idx(x, y)] === 1;
  }


  setTiles(tiles: Tile[]) {
    if (tiles.length !== this.w * this.h) {
      throw new Error("GridSystem.setTiles: invalid tiles length");
    }
    this.tiles = tiles;
  }


  clearOccupancy() {
    this.occupancy.clear();
  }


  actorIdAt(x: number, y: number): string | undefined {
    return this.occupancy.get(this.key(x, y));
  }


  setActorAt(x: number, y: number, actorId: string) {
    this.occupancy.set(this.key(x, y), actorId);
  }


  clearActorAt(x: number, y: number) {
    this.occupancy.delete(this.key(x, y));
  }


  moveActor(actorId: string, fromX: number, fromY: number, toX: number, toY: number) {
    this.clearActorAt(fromX, fromY);
    this.setActorAt(toX, toY, actorId);
  }
}

export class TurnSystem {

  turnIndex = 0;
  phase: TurnPhase = "PLAYER";

    this.actors = actors;
    this.turnIndex = 0;
    this.phase = "PLAYER";

    return this.actors[this.turnIndex];

  }

    if (this.actors.length === 0) {

      return;
    }
    if (this.turnIndex >= this.actors.length) {
      this.turnIndex = 0;
